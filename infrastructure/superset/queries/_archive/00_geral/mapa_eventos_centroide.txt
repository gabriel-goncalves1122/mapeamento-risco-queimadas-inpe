-- ============================================================
-- CHART: Deck.gl Scatterplot (Mapa Geoespacial)
-- LATITUDE: latitude_centroide
-- LONGITUDE: longitude_centroide
-- RAIO: proporcional a total_focos
-- COR: por fk_bioma
-- OBJETIVO: Mancha geoespacial dos eventos de incêndio ativos
-- DASHBOARD: Painel Geral (Visão Executiva)
-- ============================================================
SELECT
  fire_event_id,
  __time AS data_inicio,
  latitude_centroide,
  longitude_centroide,
  total_focos,
  ROUND(soma_frp, 2) AS soma_frp,
  ROUND(area_evento_km2, 2) AS area_km2,
  ROUND(duracao_dias, 1) AS duracao_dias,
  fk_bioma AS bioma,
  LOOKUP(fk_municipio_principal, 'municipios_nome') AS municipio,
  LOOKUP(fk_municipio_principal, 'municipios_uf') AS uf
FROM fato_eventos_incendio
WHERE __time >= CURRENT_TIMESTAMP - INTERVAL '7' DAY
  AND latitude_centroide IS NOT NULL
  AND longitude_centroide IS NOT NULL
ORDER BY total_focos DESC
LIMIT 500
