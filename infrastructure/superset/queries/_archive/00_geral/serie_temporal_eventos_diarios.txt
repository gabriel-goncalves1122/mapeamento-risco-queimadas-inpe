-- ============================================================
-- CHART: Line Chart (Série Temporal)
-- EIXO X: Dia
-- EIXO Y: total_eventos, total_focos
-- OBJETIVO: Evolução diária de eventos e focos na última semana
-- DASHBOARD: Painel Geral (Visão Executiva)
-- ============================================================
SELECT
  TIME_FLOOR(__time, 'P1D') AS dia,
  COUNT(*) AS total_eventos,
  SUM(total_focos) AS total_focos,
  ROUND(SUM(soma_frp), 2) AS frp_acumulada,
  ROUND(AVG(media_risco_fogo), 2) AS risco_medio
FROM fato_eventos_incendio
WHERE __time >= CURRENT_TIMESTAMP - INTERVAL '7' DAY
GROUP BY 1
ORDER BY 1
