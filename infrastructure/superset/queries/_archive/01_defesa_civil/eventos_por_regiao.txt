-- ============================================================
-- CHART: Stacked Bar Chart
-- EIXO X: regiao_brasil
-- EIXO Y: total_eventos (empilhado por bioma)
-- OBJETIVO: Eventos agrupados por região do Brasil para alocação de recursos
-- DASHBOARD: Defesa Civil & Ministério da Saúde
-- ============================================================
SELECT
  LOOKUP(fk_municipio_principal, 'municipios_regiao') AS regiao,
  fk_bioma AS bioma,
  COUNT(*) AS total_eventos,
  SUM(total_focos) AS total_focos,
  ROUND(SUM(area_evento_km2), 2) AS area_total_km2
FROM fato_eventos_incendio
WHERE __time >= CURRENT_TIMESTAMP - INTERVAL '7' DAY
  AND fk_municipio_principal IS NOT NULL
  AND fk_municipio_principal != 'NAO_INFORMADO'
GROUP BY 1, 2
ORDER BY total_eventos DESC
