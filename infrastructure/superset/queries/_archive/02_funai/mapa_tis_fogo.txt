-- ============================================================
-- CHART: Deck.gl Scatterplot (Mapa Geoespacial)
-- LATITUDE: latitude_centroide
-- LONGITUDE: longitude_centroide
-- RAIO: proporcional a total_focos
-- COR: verde = entorno, vermelho = dentro da TI
-- OBJETIVO: Mapa dos eventos que interceptam ou ameaçam TIs
-- DASHBOARD: FUNAI & ONGs Indígenas
-- ============================================================
SELECT
  fire_event_id,
  latitude_centroide,
  longitude_centroide,
  total_focos,
  ROUND(indice_pressao_terra_indigena, 2) AS pressao_ti,
  ROUND(distancia_terra_indigena_km, 2) AS distancia_ti_km,
  CASE
    WHEN fk_terra_indigena IS NOT NULL AND fk_terra_indigena != 'NAO_INFORMADO'
    THEN 'DENTRO'
    ELSE 'ENTORNO'
  END AS tipo_ameaca,
  COALESCE(
    LOOKUP(fk_terra_indigena, 'ti_nome'),
    LOOKUP(fk_terra_indigena_proxima, 'ti_nome')
  ) AS terra_indigena,
  COALESCE(
    LOOKUP(fk_terra_indigena, 'ti_etnia'),
    LOOKUP(fk_terra_indigena_proxima, 'ti_etnia')
  ) AS etnia
FROM fato_eventos_incendio
WHERE __time >= CURRENT_TIMESTAMP - INTERVAL '7' DAY
  AND latitude_centroide IS NOT NULL
  AND (
    (fk_terra_indigena IS NOT NULL AND fk_terra_indigena != 'NAO_INFORMADO')
    OR indice_pressao_terra_indigena > 0
  )
ORDER BY total_focos DESC
LIMIT 300
