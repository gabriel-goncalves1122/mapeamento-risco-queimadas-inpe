-- ============================================================
-- CHART: Big Number (KPI)
-- OBJETIVO: Quantas Terras Indígenas tiveram fogo DENTRO do polígono
-- MÉTRICA: COUNT DISTINCT de fk_terra_indigena (onde != NAO_INFORMADO)
-- DASHBOARD: FUNAI & ONGs Indígenas
-- ============================================================
SELECT
  COUNT(DISTINCT fk_terra_indigena) AS tis_invadidas
FROM fato_eventos_incendio
WHERE __time >= CURRENT_TIMESTAMP - INTERVAL '7' DAY
  AND fk_terra_indigena IS NOT NULL
  AND fk_terra_indigena != 'NAO_INFORMADO'
