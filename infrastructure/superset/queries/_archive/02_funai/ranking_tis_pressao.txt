-- ============================================================
-- CHART: Bar Chart Horizontal
-- EIXO Y: nome_terra_indigena (etnia)
-- EIXO X: pressao_acumulada
-- OBJETIVO: Top 15 TIs com maior pressão acumulada (fogo no entorno)
-- DASHBOARD: FUNAI & ONGs Indígenas
-- ============================================================
SELECT
  LOOKUP(fk_terra_indigena_proxima, 'ti_nome') AS terra_indigena,
  LOOKUP(fk_terra_indigena_proxima, 'ti_etnia') AS etnia,
  LOOKUP(fk_terra_indigena_proxima, 'ti_uf') AS uf,
  COUNT(*) AS total_eventos,
  SUM(total_focos) AS total_focos,
  ROUND(SUM(indice_pressao_terra_indigena), 2) AS pressao_acumulada
FROM fato_eventos_incendio
WHERE __time >= CURRENT_TIMESTAMP - INTERVAL '7' DAY
  AND fk_terra_indigena_proxima IS NOT NULL
  AND fk_terra_indigena_proxima != 'NAO_INFORMADO'
GROUP BY 1, 2, 3
ORDER BY pressao_acumulada DESC
LIMIT 15
